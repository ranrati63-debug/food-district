RewriteEngine On

# Redirect order.html to /order
RewriteCond %{THE_REQUEST} \s/order\.html[\s?] [NC]
RewriteRule ^order\.html$ https://fooddistrict.in/order [R=301,L]

# Internally load order.html
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME}.html -f
RewriteRule ^(.+?)/?$ $1.html [L]
